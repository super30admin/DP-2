class Solution {
    public int minCost(int[][] costs) {
        int[][] minCost = new int[costs.length][3];

        //setup for the last row of the matrix as base to buildupon 
        for(int i=0 ; i < 3 ;i++){
            minCost[costs.length-1][i] = costs[costs.length-1][i]; 
        }

        //Bottom up way of calculating the n-2th level based on its below levels minimum & propogating up top to 0th index

        for(int i = costs.length-2; i >= 0 ; i --){
            int rCost = costs[i][0] + Math.min(minCost[i+1][1], minCost[i+1][2]); 
            int bCost = costs[i][1] + Math.min(minCost[i+1][0], minCost[i+1][2]);
            int gCost = costs[i][2] + Math.min(minCost[i+1][0], minCost[i+1][1]);
            minCost[i][0] = rCost;
            minCost[i][1] = bCost;
            minCost[i][2] = gCost;
        }
        return Math.min(minCost[0][0], Math.min(minCost[0][1],minCost[0][2]));
    }
}
